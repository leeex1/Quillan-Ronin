cmake_minimum_required(VERSION 3.20)
project(NextVerse)

set(CMAKE_CXX_STANDARD 17)
find_package(Boost REQUIRED COMPONENTS system)
find_package(SDL2 REQUIRED)
find_package(ZMQ REQUIRED)
find_package(JUCE REQUIRED)

add_executable(nextverse
    main.cpp
    quantum_core.hpp
    quantum_vm.hpp
    game_engine.hpp
    mmorpg.hpp
    daw.hpp
    network.hpp
)

target_include_directories(nextverse PRIVATE
    ${Boost_INCLUDE_DIRS}
    ${SDL2_INCLUDE_DIRS}
    ${ZMQ_INCLUDE_DIRS}
    ${JUCE_INCLUDE_DIRS}
)

target_link_libraries(nextverse PRIVATE
    ${Boost_LIBRARIES}
    ${SDL2_LIBRARIES}
    ${ZMQ_LIBRARIES}
    ${JUCE_LIBRARIES}
)
